cmake_minimum_required(VERSION 3.20)
project(HappyWheelRagdoll LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
set(sdk "$ENV{HL2SDKCS2}/public")
set(mm "$ENV{MMSOURCE_DEV}")
add_library(HappyWheelNative SHARED runtime.cpp pose.cpp)
target_include_directories(HappyWheelNative PRIVATE "${mm}/core" "${mm}/core/sourcehook" "${mm}/third_party/khook/include" "${sdk}" "${sdk}/mathlib" "${sdk}/tier0" "${sdk}/tier1")
target_compile_definitions(HappyWheelNative PRIVATE META_IS_SOURCE2 COMPILER_MSVC COMPILER_MSVC64 PLATFORM_64BITS X64BITS NOMINMAX _CRT_SECURE_NO_WARNINGS)
target_link_libraries(HappyWheelNative PRIVATE "$ENV{HL2SDKCS2}/lib/public/win64/tier0.lib")
add_executable(CoreTests tests/core_tests.cpp)
target_include_directories(CoreTests PRIVATE core)
enable_testing()
add_test(NAME CoreTests COMMAND CoreTests)
set(hl "$ENV{HL2SDKCS2}")
set(proto "${hl}/thirdparty/protobuf-3.21.8/src")
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/network_connection.pb.h"
    COMMAND "${hl}/devtools/bin/protoc.exe" "--cpp_out=${CMAKE_CURRENT_BINARY_DIR}" "--proto_path=${hl}/common" "--proto_path=${proto}" network_connection.proto
    DEPENDS "${hl}/common/network_connection.proto" VERBATIM)
target_sources(HappyWheelNative PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/network_connection.pb.h")
target_include_directories(HappyWheelNative PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${proto}" "${sdk}/engine")
target_link_libraries(HappyWheelNative PRIVATE "${hl}/lib/public/win64/interfaces.lib")
